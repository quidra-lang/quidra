#include <iostream>
#include <string>

int main() {
    long long state = 7;
    long long s = 0, m = 0, e = 0;
    std::string j;
    for (int k = 0; k < 50; ++k) {
        state = (state * 48271) % 2147483647;
        long long t = state % 1000;
        s += t;
        if (t > m) m = t;
        if (t % 2 == 0) ++e;
        if (k < 5) { if (k) j += "-"; j += std::to_string(t); }
    }
    std::cout << "SUM " << s << "\n" << "MAX " << m << "\n"
              << "EVENS " << e << "\n" << "JOINED " << j << "\n";
    return 0;
}
