#!/usr/bin/env python3
"""MB-02 -- Factorial: recompute k! mod M from scratch for every k.

Frozen workload definition: methodology/06_micro_workloads.md section 4, MB-02.
The inner loop must restart from f = 1 for every k; no partial-factorial reuse.
"""

import sys
import time

M = 1000003
N = 20000


def workload(M=M, N=N):
    # The two pinned constants are bound as function locals (the ordinary
    # CPython idiom for a hot loop) so that the inner statement does not pay a
    # global lookup that every compiled configuration folds into its
    # instruction stream. Values, loop structure and arithmetic are unchanged.
    total = 0
    for k in range(1, N + 1):
        f = 1
        for j in range(2, k + 1):
            f = (f * j) % M
        total = (total + f) % M
    return total


def main():
    mode = sys.argv[1] if len(sys.argv) > 1 else "once"
    if mode == "steady":
        k_count = int(sys.argv[2]) if len(sys.argv) > 2 else 7
        for k in range(k_count):
            start = time.perf_counter_ns()
            total = workload()
            elapsed = time.perf_counter_ns() - start
            print("ITER %d %d" % (k, elapsed), flush=True)
    else:
        total = workload()
    print("MB02 %d" % total)


if __name__ == "__main__":
    main()
