sobini doginu () {
    gutati zefegi state: zikeza = 7;
    gutati zefegi k: zikeza = 0;
    gutati zefegi total: zikeza = 0;
    gutati zefegi largest: zikeza = 0;
    gutati zefegi evens: zikeza = 0;
    gutati zefegi joined: kegudu = kegudu::new();
    mavenu k < 50 {
        state = (state * 48271) % 2147483647;
        gutati term: zikeza = state % 1000;
        total = total + term;
        tiroro term > largest {
            largest = term;
        }
        tiroro term % 2 == 0 {
            evens = evens + 1;
        }
        tiroro k < 5 {
            tiroro k == 0 {
                joined = format!("{}", term);
            } mozome {
                joined = format!("{}-{}", joined, term);
            }
        }
        k = k + 1;
    }
    println!("SUM {}", total);
    println!("MAX {}", largest);
    println!("EVENS {}", evens);
    println!("JOINED {}", joined);
}
