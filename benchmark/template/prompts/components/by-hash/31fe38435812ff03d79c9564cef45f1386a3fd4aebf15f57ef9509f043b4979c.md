#!/usr/bin/env python3
"""MB-07 -- Sorting: bottom-up iterative merge sort, N = 2000000, R = 4.

Frozen workload definition: methodology/06_micro_workloads.md section 4, MB-07.
The pinned algorithm only: ascending, stable, ping-pong buffers. No `sorted()`,
no `list.sort()`, no insertion-sort cut-off, no early exit, no radix variant.
The per-round copy of `src` into `a` is part of the workload.
Data container: `array('q')` -- the 64-bit integer array pinned for the
`python` configuration by the binding table in section 4.12(a). Not the
implementer's choice.
"""

import sys
import time

from array import array

N = 2000000
R = 4
SEED = 20267917


class Lehmer:
    """Park-Miller minimal-standard LCG: state = (48271 * state) mod 2147483647."""

    __slots__ = ("state",)

    def __init__(self, seed):
        self.state = seed

    def next_int(self):
        self.state = (48271 * self.state) % 2147483647
        return self.state


def msort(a, buf, n):
    """Bottom-up merge sort of a[0:n]; buf is the ping-pong partner buffer."""
    src = a
    dst = buf
    width = 1
    while width < n:
        lo = 0
        while lo < n:
            mid = min(lo + width, n)
            hi = min(lo + 2 * width, n)
            i = lo
            j = mid
            k = lo
            while i < mid and j < hi:
                if src[i] <= src[j]:
                    dst[k] = src[i]
                    i = i + 1
                else:
                    dst[k] = src[j]
                    j = j + 1
                k = k + 1
            while i < mid:
                dst[k] = src[i]
                i = i + 1
                k = k + 1
            while j < hi:
                dst[k] = src[j]
                j = j + 1
                k = k + 1
            lo = lo + 2 * width
        src, dst = dst, src
        width = width * 2
    if src is not a:
        a[:] = src


def workload():
    g = Lehmer(SEED)
    src = array("q", [g.next_int() for _ in range(N)])
    buf = array("q", bytes(8 * N))

    total = 0
    ssum = 0
    inv = 0
    for r in range(R):
        src[r] = src[r] + 1
        a = array("q", src)
        msort(a, buf, N)

        chk = 0
        for i in range(N):
            chk = (chk * 31 + a[i] % 1000003) % 1000003
        total = (total * 7 + chk) % 1000003

        ssum = 0
        for i in range(N):
            ssum = ssum + a[i]

        for i in range(1, N):
            if a[i - 1] > a[i]:
                inv = inv + 1
    return total, ssum, inv


def main():
    mode = sys.argv[1] if len(sys.argv) > 1 else "once"
    if mode == "steady":
        k_count = int(sys.argv[2]) if len(sys.argv) > 2 else 7
        for k in range(k_count):
            start = time.perf_counter_ns()
            total, ssum, inv = workload()
            elapsed = time.perf_counter_ns() - start
            print("ITER %d %d" % (k, elapsed), flush=True)
    else:
        total, ssum, inv = workload()
    print("MB07 %d %d %d" % (total, ssum, inv))


if __name__ == "__main__":
    main()
