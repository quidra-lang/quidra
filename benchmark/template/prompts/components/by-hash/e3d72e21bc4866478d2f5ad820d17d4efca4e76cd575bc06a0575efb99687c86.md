// MB-09 -- statistics: two-pass moments, Pearson correlation, histogram.
#include <array>
#include <chrono>
#include <cmath>
#include <cstdint>
#include <cstdlib>
#include <cstdio>
#include <iostream>

namespace {

constexpr std::int64_t kModulus = 2147483647;
constexpr std::int64_t kMultiplier = 48271;

// Lehmer / MINSTD generator: s = (48271 * s) mod (2^31 - 1).
class Lcg {
public:
    explicit Lcg(std::int64_t seed) : state_(seed) {}

    std::int64_t next_int() {
        state_ = (kMultiplier * state_) % kModulus;
        return state_;
    }

    double next_unit() { return static_cast<double>(next_int()) / 2147483647.0; }

private:
    std::int64_t state_;
};

constexpr std::size_t kN = 2000000;
constexpr int kRounds = 30;
constexpr std::int64_t kSeed = 20269917;

std::string run() {
    Lcg g(kSeed);
    std::vector<double> x(kN);
    std::vector<double> y(kN);
    for (std::size_t i = 0; i < kN; ++i) {
        x[i] = g.next_unit() * 100.0;
    }
    for (std::size_t i = 0; i < kN; ++i) {
        y[i] = g.next_unit() * 100.0;
    }

    double mean = 0.0;
    double var = 0.0;
    double sd = 0.0;
    double mn = 0.0;
    double mx = 0.0;
    double mad = 0.0;
    double pearson = 0.0;
    std::int64_t hist_chk = 0;

    for (int r = 0; r < kRounds; ++r) {
        x[static_cast<std::size_t>(r)] += 1.0e-9;  // anti-elimination

        double s = 0.0;  // pass 1
        mn = x[0];
        mx = x[0];
        for (std::size_t i = 0; i < kN; ++i) {
            const double v = x[i];
            s = s + v;
            if (v < mn) {
                mn = v;
            }
            if (v > mx) {
                mx = v;
            }
        }
        mean = s / static_cast<double>(kN);

        double sq = 0.0;  // pass 2
        double ad = 0.0;
        for (std::size_t i = 0; i < kN; ++i) {
            const double d = x[i] - mean;
            sq = sq + d * d;
            ad = ad + (d < 0 ? -d : d);
        }
        var = sq / static_cast<double>(kN);
        sd = std::sqrt(var);
        mad = ad / static_cast<double>(kN);

        double sy = 0.0;  // pass 3
        for (std::size_t i = 0; i < kN; ++i) {
            sy = sy + y[i];
        }
        const double meany = sy / static_cast<double>(kN);

        double sxy = 0.0;  // pass 4
        double sxx = 0.0;
        double syy = 0.0;
        for (std::size_t i = 0; i < kN; ++i) {
            const double dx = x[i] - mean;
            const double dy = y[i] - meany;
            sxy = sxy + dx * dy;
            sxx = sxx + dx * dx;
            syy = syy + dy * dy;
        }
        pearson = sxy / std::sqrt(sxx * syy);

        std::array<std::int64_t, 64> hist{};  // pass 5
        for (std::size_t i = 0; i < kN; ++i) {
            int b = static_cast<int>(std::floor(x[i] * 0.64));
            if (b < 0) {
                b = 0;
            }
            if (b > 63) {
                b = 63;
            }
            hist[static_cast<std::size_t>(b)] += 1;
        }
        hist_chk = 0;
        for (int b = 0; b < 64; ++b) {
            hist_chk += (b + 1) * hist[static_cast<std::size_t>(b)];
        }
    }

    char buffer[512];
    std::snprintf(buffer, sizeof(buffer),
                  "MB09 mean=%.16e var=%.16e sd=%.16e min=%.16e max=%.16e mad=%.16e "
                  "pearson=%.16e hist_chk=%lld",
                  mean, var, sd, mn, mx, mad, pearson, static_cast<long long>(hist_chk));
    return std::string(buffer);
}

}  // namespace
