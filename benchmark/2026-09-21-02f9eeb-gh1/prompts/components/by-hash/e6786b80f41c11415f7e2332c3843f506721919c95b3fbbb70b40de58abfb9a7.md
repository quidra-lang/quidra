import time

N = 120000000
SEED = 20263917

    s_xor = 0
    s_mul = 1
    s_div = 0
    for _ in range(N):

        s_add = (s_add + x) % 2147483647
        s_xor = s_xor ^ x
        s_mul = (s_mul * 33 + (x % 97)) % 1000003
