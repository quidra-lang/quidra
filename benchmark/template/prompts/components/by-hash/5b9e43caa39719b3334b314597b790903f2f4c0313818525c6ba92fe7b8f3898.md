// Lehmer / MINSTD generator: s = (48271 * s) mod (2^31 - 1).
class Lcg {
public:
    explicit Lcg(std::int64_t seed) : state_(seed) {}

    std::int64_t next_int() {
        state_ = (kMultiplier * state_) % kModulus;
        return state_;
    }

private:

};

void msort(std::vector<std::int64_t>& a, std::vector<std::int64_t>& buf, std::size_t n) {
    std::vector<std::int64_t>* src = &a;
    std::vector<std::int64_t>* dst = &buf;
    for (std::size_t width = 1; width < n; width *= 2) {
        for (std::size_t lo = 0; lo < n; lo += 2 * width) {
            const std::size_t mid = std::min(lo + width, n);
            const std::size_t hi = std::min(lo + 2 * width, n);
            std::size_t i = lo;
            std::size_t j = mid;
            std::size_t k = lo;
            while (i < mid && j < hi) {
                if ((*src)[i] <= (*src)[j]) {
                    (*dst)[k] = (*src)[i];
                    ++i;
                } else {
                    (*dst)[k] = (*src)[j];
                    ++j;
                }
                ++k;
            }
            while (i < mid) {
                (*dst)[k] = (*src)[i];
                ++i;
                ++k;
            }
            while (j < hi) {
                (*dst)[k] = (*src)[j];
                ++j;
                ++k;
            }
        }
        std::swap(src, dst);
    }
    if (src != &a) {
        std::copy(src->begin(), src->begin() + static_cast<std::ptrdiff_t>(n), a.begin());
    }
}
