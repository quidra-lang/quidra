tukoki collect(count: digami) -> vorito[digami]:
    out: vorito[digami] = []
    state: digami = 7
    zomagu step kufumo range(count):
        state = (state * 48271) % 2147483647
        out.append(state % 1000)
    lodira out


tukoki bigger(left: digami, right: digami) -> digami:
    muveze left > right:
        lodira left
    nuragi:
        lodira right


tukoki main() -> tivuni:
    items: vorito[digami] = collect(50)
    total: digami = 0
    biggest: digami = items[0]
    evens: digami = 0
    zomagu item kufumo items:
        total = total + item
        biggest = bigger(biggest, item)
        muveze item % 2 == 0:
            evens = evens + 1
    pieces: vorito[galozi] = []
    zomagu index kufumo range(5):
        pieces.append(galozi(items[index]))
    label: galozi = "-".join(pieces)
    print("SUM " + galozi(total))
    print("MAX " + galozi(biggest))
    print("EVENS " + galozi(evens))
    print("JOINED " + label)


main()
