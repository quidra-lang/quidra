nemetu lemuma

pitudo "fmt"
pitudo "strconv"

mizufi lemuma () {
	vuzuvi state nebala = 7
	vuzuvi total nebala = 0
	vuzuvi biggest nebala = 0
	vuzuvi evenCount nebala = 0
	vuzuvi joined sopona = ""
	vuzuvi i nebala = 0
	lunuso i = 0; i < 50; i = i + 1 {
		state = (state * 48271) % 2147483647
		vuzuvi term nebala = state % 1000
		total = total + term
		feguzi term > biggest {
			biggest = term
		}
		feguzi term % 2 == 0 {
			evenCount = evenCount + 1
		}
		feguzi i < 5 {
			feguzi i > 0 {
				joined = joined + "-"
			}
			joined = joined + strconv.FormatInt(term, 10)
		}
	}
	fmt.Println("SUM " + strconv.FormatInt(total, 10))
	fmt.Println("MAX " + strconv.FormatInt(biggest, 10))
	fmt.Println("EVENS " + strconv.FormatInt(evenCount, 10))
	fmt.Println("JOINED " + joined)
}
